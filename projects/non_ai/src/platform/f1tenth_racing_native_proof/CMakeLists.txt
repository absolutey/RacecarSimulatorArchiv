cmake_minimum_required(VERSION 3.8)
project(f1tenth_racing_native_proof)
find_package(ament_cmake REQUIRED)
# V7: bind the ROS native build to the unchanged independent-stop implementation.
# Refuse silent forked C++ proof implementations inside this source workspace.
set(AUTHORITATIVE_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../environment/f1tenth_racing_sim/f1tenth_racing_sim/safety/native_stop_proof.cpp")
if(NOT EXISTS "${AUTHORITATIVE_CPP}")
  message(FATAL_ERROR "NATIVE_PROOF_REFERENCE_SOURCE_MISSING")
endif()
file(SHA256 "${AUTHORITATIVE_CPP}" reference_cpp_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/native_stop_proof.cpp" native_cpp_sha)
if(NOT reference_cpp_sha STREQUAL native_cpp_sha)
  message(FATAL_ERROR "NATIVE_PROOF_CPP_SOURCE_DRIFT_BLOCKED")
endif()
add_library(f1tenth_native_stop SHARED src/native_stop_proof.cpp)
target_compile_features(f1tenth_native_stop PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(f1tenth_native_stop PRIVATE -O3 -fno-fast-math -Wall -Wextra -Werror=return-type)
endif()
# Explicit SOLO-only C++ ROS Guard implementation. Not the default publisher.
find_package(rclcpp REQUIRED)
find_package(control_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(native_actuator_guard src/native_guard_ros.cpp)
target_compile_features(native_actuator_guard PRIVATE cxx_std_17)
ament_target_dependencies(native_actuator_guard rclcpp control_msgs ackermann_msgs std_msgs builtin_interfaces)
target_link_libraries(native_actuator_guard f1tenth_native_stop OpenSSL::Crypto)
install(TARGETS native_actuator_guard RUNTIME DESTINATION lib/${PROJECT_NAME})
install(FILES CREATE_NATIVE_GUARD_LAUNCH_PARAMS.py DESTINATION share/${PROJECT_NAME})
install(TARGETS f1tenth_native_stop LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES src/native_stop_proof.cpp DESTINATION share/${PROJECT_NAME}/source)
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(test_native_binary_contract test/test_native_binary_contract.py
   ENV "F1_NATIVE_LIBRARY=$<TARGET_FILE:f1tenth_native_stop>"
              "F1_NATIVE_IMPLEMENTATION_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}/src/native_stop_proof.cpp"
              "F1_NATIVE_REFERENCE_SOURCE=${AUTHORITATIVE_CPP}")
 ament_add_pytest_test(test_native_guard_humble_api_secondpass test/test_native_guard_humble_api_secondpass.py)
 ament_add_pytest_test(test_native_guard_identity_thirdpass test/test_native_guard_identity_thirdpass.py)
 ament_add_pytest_test(test_native_guard_bounds_fourlayer test/test_native_guard_bounds_fourlayer.py)
endif()
ament_package()
